from pathlib import Path
import json,subprocess,hashlib,time,sys
binary=sys.argv[1];output=Path(sys.argv[2]);output.mkdir(parents=True,exist_ok=False)
root=Path('/tmp/bateia0062-gate');fixtures=json.loads((root/'fixtures.json').read_text());rows=[]
for f in fixtures:
 path=root/f['path'];receipt=output/(f['name']+'.json');argv=[binary,'effects','verify','--source',str(path),'--symbol',f['symbol'],'--seed','1','--timeout','1s','--output',str(receipt)];start=time.monotonic()
 r=subprocess.run(argv,capture_output=True,text=True,timeout=90)
 (output/(f['name']+'.stdout')).write_text(r.stdout);(output/(f['name']+'.stderr')).write_text(r.stderr)
 row={'name':f['name'],'argv':argv,'source_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'binary_sha256':hashlib.sha256(Path(binary).read_bytes()).hexdigest(),'exit_code':r.returncode,'telemetry_seconds':time.monotonic()-start}
 if receipt.exists():
  v=subprocess.run(['python3',str(root/'oracle.py'),'--report',str(receipt),'--expected',f['expected']],capture_output=True,text=True)
  row.update({'oracle_exit':v.returncode,'oracle_stdout':v.stdout,'oracle_stderr':v.stderr,'receipt_sha256':hashlib.sha256(receipt.read_bytes()).hexdigest()})
 rows.append(row);(output/'verification.json').write_text(json.dumps(rows,indent=2)+'\n');print(f['name'],r.returncode,row.get('oracle_exit'),flush=True)
